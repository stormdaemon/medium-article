 }
  ctx.restore();
}

function draw(){
  const track=tracks[raceIndex]||tracks[0];
  ctx.clearRect(0,0,W,H);
  drawSky(track);
  if(player){
    drawBuildings(track);
    drawRoad(track);
    drawWorldCars(track);
    drawPlayer(track);
    drawSpeedLines();
    comicTexture(.035);
    if(flash>0){ctx.fillStyle=`rgba(255,255,255,${flash*.28})`;ctx.fillRect(0,0,W,H);}
  }else{
    ctx.fillStyle='#ffd62e';ctx.fillRect(0,H*.35,W,H*.65);comicTexture(.08);
  }
}

function frame(t){
  const dt=Math.min(.033,(t-lastT)/1000||0);lastT=t;
  update(dt);draw();requestAnimationFrame(frame);
}
requestAnimationFrame(frame);
