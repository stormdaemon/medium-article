function initAudio(){
  if(!soundOn)return;
  if(!audioCtx) audioCtx=new (window.AudioContext||window.webkitAudioContext)();
  if(audioCtx.state==='suspended') audioCtx.resume();
}
function blip(freq=120,dur=.08,type='square',vol=.05){
  if(!soundOn)return;
  initAudio();
  const o=audioCtx.createOscillator(),g=audioCtx.createGain();
  o.type=type;o.frequency.value=freq;g.gain.value=vol;
  o.connect(g);g.connect(audioCtx.destination);o.start();
  g.gain.exponentialRampToValueAtTime(.0001,audioCtx.currentTime+dur);
  o.stop(audioCtx.currentTime+dur);
}

function seeded(n){return Math.abs(Math.sin(n*12.9898+78.233)*43758.5453)%1}
function curveAt(z,track){
  const s=track.seed;
  return (
    Math.sin(z/720+s)*.58 +
    Math.sin(z/1430+s*1.7)*.72 +
    Math.sin(z/310+s*.31)*.12
  )*track.curve;
}

function pointsForPlace(p){return [12,10,8,7,6,5,4,3,2,1,0,0][p-1]||0}

function finishRace(){
  if(state!=='race')return;
  state='results';
  hud.classList.remove('show');controls.classList.remove('show');
  const now=(performance.now()-raceStartAt)/1000;
  const all=[{driver:selected,dist:player.dist,finishTime:now,me:true},...racers.map(r=>({
    driver:r.driver,dist:r.dist,finishTime:r.finishTime,me:false
  }))];
  all.sort((a,b)=>{
    const af=a.finishTime??9999,bf=b.finishTime??9999;
    if(a.finishTime!=null||b.finishTime!=null)return af-bf;
    return b.dist-a.dist;
  });
  raceResult=all;
  all.forEach((r,idx)=>{
    const s=standings.find(x=>x.i===r.driver);
    s.pts+=pointsForPlace(idx+1);
  });
  setTimeout(()=>showResults(),450);
}

  results.classList.add('show');
  const myPlace=raceResult.findIndex(r=>r.driver===selected)+1;
  document.getElementById('resultTrack').textContent=`${tracks[raceIndex].name} · COURSE ${raceIndex+1}/5`;
  document.getElementById('placeText').textContent=myPlace===1?'1ER':myPlace+'E';
  document.getElementById('gpDots').innerHTML=tracks.map((_,i)=>`<i class="${i<=raceIndex?'done':''}"></i>`).join('');
  const sortedGP=[...standings].sort((a,b)=>b.pts-a.pts);
  const tbody=document.getElementById('resultsBody');tbody.innerHTML='';
  raceResult.forEach((r,idx)=>{
    const tr=document.createElement('tr'); if(r.driver===selected)tr.className='me';
    const gp=sortedGP.findIndex(s=>s.i===r.driver)+1;
    tr.innerHTML=`<td>${idx+1}</td><td>${drivers[r.driver].face} ${drivers[r.driver].name}</td><td>+${pointsForPlace(idx+1)}</td><td>${standings.find(s=>s.i===r.driver).pts} pts · #${gp}</td>`;
    tbody.appendChild(tr);
  });
  document.getElementById('nextBtn').textContent=raceIndex===tracks.length-1?'Classement final':'Course suivante';
}
