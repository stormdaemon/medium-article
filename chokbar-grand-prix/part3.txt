0))+'px';el.style.top=Math.max(70,y-60)+'px';
  impactLayer.appendChild(el);setTimeout(()=>el.remove(),760);
}

function positionNow(){
  const list=[{driver:selected,dist:player.dist},...racers.map(r=>({driver:r.driver,dist:r.dist}))].sort((a,b)=>b.dist-a.dist);
  return list.findIndex(r=>r.driver===selected)+1;
}

function update(dt){
  if(state!=='race')return;
  const track=tracks[raceIndex];
  collisionCooldown=Math.max(0,collisionCooldown-dt);
  shake=Math.max(0,shake-dt*2.7);flash=Math.max(0,flash-dt*3.5);

  const steering=(input.right?1:0)-(input.left?1:0);
  const targetBase=214*player.spd;
  const boosting=input.boost&&player.boost>0;
  if(boosting){
    player.boost=Math.max(0,player.boost-33*dt);
    player.speed += (356-targetBase)*dt*2.7;
    if(Math.random()<dt*7)boostSound();
  }else{
    player.speed += (targetBase-player.speed)*dt*2.3;
  }
  player.speed=Math.max(0,Math.min(372,player.speed));
  player.x += steering*dt*1.55*player.turn*(boosting?.86:1);
  player.x *= Math.pow(.998,dt*60);
  if(Math.abs(player.x)>.87){
    player.x=Math.sign(player.x)*.87;
    player.speed*=Math.pow(.965,dt*60);
    if(Math.random()<dt*5)showImpact('SKRR!',W*(.5+player.x*.38),H*.72);
  }
  player.dist += player.speed*dt;

  racers.forEach((r,idx)=>{
    r.laneTimer-=dt;
    if(r.laneTimer<=0){
      r.laneTimer=1.3+seeded(r.driver*7+r.dist*.001)*2.6;
      let lanes=[-.58,-.2,.2,.58];
      r.targetX=lanes[Math.floor(seeded(r.dist*.013+r.driver)*lanes.length)];
    }
    r.x += (r.targetX-r.x)*dt*(.55+seeded(r.driver)*.4);
    const rubber=(player.dist-r.dist)*.0022;
    const racePressure=Math.max(-8,Math.min(9,rubber));
    r.speed += ((r.baseSpeed+racePressure)-r.speed)*dt*.6;
    r.dist+=r.speed*dt;
    if(!r.finished&&r.dist>=track.length){
      r.finished=true;r.finishTime=(performance.now()-raceStartAt)/1000;
    }
  });

  traffic.forEach((t,i)=>{
    t.hit=Math.max(0,t.hit-dt);
    t.dist+=t.speed*dt;
    if(t.dist<player.dist-300)t.dist+=track.length*.72;
    if(t.dist>player.dist+track.length*.8)t.dist-=track.length*.72;
  });

  if(collisionCooldown<=0){
    let victim=null;
    for(const r of racers){
      const dz=r.dist-player.dist;
      if(Math.abs(dz)<34&&Math.abs(r.x-player.x)<.24){victim=r;break;}
    }
    if(victim){
      collisionCooldown=.72;
      const dir=(victim.x-player.x)||((Math.random()>.5)?1:-1);
      victim.x+=Math.sign(dir)*.38;
      victim.speed*=.88;
      player.speed*=.91;
      player.boost=Math.min(100,player.boost+24);
      shake=.55;flash=.33;
      showImpact(['BAM!','KRAK!','POW!','CHOK!'][Math.floor(Math.random()*4)],W*.5+dir*90,H*.55);
      impactSound();
    } else {
      for(const t of traffic){
        const dz=t.dist-player.dist;
        if(t.hit<=0&&Math.abs(dz)<30&&Math.abs(t.x-player.x)<.23){
          t.hit=.9;collisionCooldown=.62;
          t.x+=Math.sign((t.x-player.x)||1)*.44;
          player.speed*=.86;
          player.boost=Math.min(100,player.boost+19);
          shake=.65;flash=.4;
          showImpact(['BANG!','CLANG!','PAM!'][Math.floor(Math.random()*3)],W*.5,H*.57);
          impactSound();break;
        }
      }
    }
  }

  if(player.dist>=track.length && player.finishTime==null){
    player.finishTime=(performance.now()-raceStartAt)/1000;
    player.dist=track.length;
    finishRace();
  }

  document.getElementById('posHud').textContent=positionNow()+'/12';
  document.getElementById('speedHud').textContent=Math.round(player.speed*1.28);
  document.getElementById('progressHud').textContent=Math.min(100,Math.floor(player.dist/track.length*100))+'%';
  document.getElementById('boostFill').style.width=player.boost+'%';
}

function roadCenterFor(depth,dist,track){
  const farDist=(1-depth)*1050;
  const here=curveAt(dist,track);
  const ahead=curveAt(dist+farDist,track);
  return W*.5+(ahead-here)*W*.13*(1-depth);
}

function comicTexture(alpha=.05){
  ctx.save();ctx.globalAlpha=alpha;ctx.fillStyle='#111';
  const step=8;
  for(let y=0;y<H;y+=step){
    for(let x=((y/step)%2)*4;x<W;x+=step){
      ctx.beginPath();ctx.arc(x,y,1.0,0,Math.PI*2);ctx.fill();
    }
  }
  ctx.restore();
}

  ctx.fillStyle=track.sky;ctx.fillRect(0,0,W,H*.35);
  const sunX=W*.78,sunY=H*.13,r=Math.min(W,H)*.07;
  ctx.fillStyle='#fff4a6';ctx.beginPath();ctx.arc(sunX,sunY,r,0,Math.PI*2);ctx.fill();
  ctx.strokeStyle='#111';ctx.lineWidth=3;ctx.stroke();
  ctx.globalAlpha=.14;ctx.strokeStyle='#111';ctx.lineWidth=2;
  for(let i=0;i<14;i++){
    ctx.beginPath();ctx.moveTo(sunX,sunY);ctx.lineTo(sunX+Math.cos(i/14*Math.PI*2)*r*2.4,sunY+Math.sin(i/14*Math.PI*2)*r*2.4);ctx.stroke();
  }
  ctx.globalAlpha=1;
}

  ctx.strokeStyle='#111';ctx.lineWidth=4;

}
