    function canvasTexture(size, draw, repeatX=1, repeatY=1, srgb=true){
      const c = document.createElement('canvas'); c.width=c.height=size;
      const ctx=c.getContext('2d',{willReadFrequently:false});
      draw(ctx,size);
      const t=new THREE.CanvasTexture(c);
      if(srgb) t.colorSpace=THREE.SRGBColorSpace;
      t.wrapS=t.wrapT=THREE.RepeatWrapping;
      t.repeat.set(repeatX,repeatY);
      t.anisotropy=Math.min(maxAniso,8);
      t.needsUpdate=true;
      return t;
    }

    function makeBrickTexture(bump=false){
      return canvasTexture(768,(ctx,s)=>{
        ctx.fillStyle=bump?'#777':'#b57b49';ctx.fillRect(0,0,s,s);
        const bh=42,bw=96;
        for(let row=0;row<Math.ceil(s/bh);row++){
          const off=(row%2)*bw*.5;
          for(let x=-bw;x<s+bw;x+=bw){
            const xx=x+off,y=row*bh;
            const g=ctx.createLinearGradient(xx,y,xx,y+bh);
            if(bump){g.addColorStop(0,'#999');g.addColorStop(.18,'#8b8b8b');g.addColorStop(1,'#686868');}
            else {g.addColorStop(0,'#c68b56');g.addColorStop(.28,'#b47a49');g.addColorStop(1,'#956239');}
