    function makeJar(radius=.35,height=.8,variant=0){
      const pts=[];const count=18;
      for(let i=0;i<count;i++){const y=i/(count-1)*height;const t=y/height;let r;
        if(t<.12)r=radius*(.45+t*3.5);else if(t<.62)r=radius*(1-.18*(t-.12));else if(t<.86)r=radius*(.9-(t-.62)*2.05);else r=radius*(.38+(t-.86)*.5);
        pts.push(new THREE.Vector2(Math.max(.06,r),y));}
      const material=variant%3===0?mat.pottery:new THREE.MeshStandardMaterial({color:variant%3===1?0x8d4a32:0xc37845,roughness:.88,map:variant%2?null:tex.pottery});
      const jar=mesh(new THREE.LatheGeometry(pts,16),material);jar.castShadow=true;
      const lip=mesh(new THREE.TorusGeometry(radius*.28,.055,7,18),material,0,height,0);lip.rotation.x=Math.PI/2;jar.add(lip);

      return jar;
    }
