    function resolveCollisions(pos){
      const r=.35;for(const c of colliders){if(pos.x>c.minX-r&&pos.x<c.maxX+r&&pos.z>c.minZ-r&&pos.z<c.maxZ+r){const dl=Math.abs(pos.x-(c.minX-r)),dr=Math.abs((c.maxX+r)-pos.x),dt=Math.abs(pos.z-(c.minZ-r)),db=Math.abs((c.maxZ+r)-pos.z);const m=Math.min(dl,dr,dt,db);if(m===dl)pos.x=c.minX-r;else if(m===dr)pos.x=c.maxX+r;else if(m===dt)pos.z=c.minZ-r;else pos.z=c.maxZ+r;}}
      pos.x=clamp(pos.x,-65,65);pos.z=clamp(pos.z,-76,75);
    }

      const targetSpeed=state.sprint?8.3:4.2;state.velocity.x=lerp(state.velocity.x,move.x*targetSpeed,1-Math.exp(-dt*10));state.velocity.z=lerp(state.velocity.z,move.z*targetSpeed,1-Math.exp(-dt*10));
      const next=camera.position.clone().addScaledVector(state.velocity,dt);resolveCollisions(next);camera.position.x=next.x;camera.position.z=next.z;

    }
